2. Γραμματική της Γλώσσας σε BNF

<program> ::= <statement_list>

<statement_list> ::= <statement>
                 | <statement_list> <statement>

<statement> ::= <create_stmt> ';'
                 | <select_stmt> ';'

-- CREATE TABLE --
<create_stmt> ::= CREATE TABLE identifier '(' <column_list> ')'

<column_list> ::= <column_def>
                 | <column_list> ',' <column_def>

<column_def> ::= identifier <data_type>

<data_type> ::= INT
                 | FLOAT
                 | VARCHAR '(' int_literal ')'

-- SELECT --
<select_stmt> ::= SELECT <col_select>
                    FROM <table_ref>
                    <join_clauses>
                    <opt_where>
                    <opt_group>
                    <opt_order>
                    <opt_limit>

<col_select> ::= '*'
                 | <col_ref_list>

<col_ref_list> ::= <col_ref>
                 | <col_ref_list> ',' <col_ref>

<col_ref> ::= identifier
                 | identifier '.' identifier

<table_ref> ::= identifier
                 | identifier AS identifier

-- JOIN (Ερώτημα 3) --
<join_clauses> ::= ε
                 | <join_clauses> <join_clause>

<join_clause> ::= JOIN <table_ref> ON <qualified_col> '=' <qualified_col>

<qualified_col> ::= identifier '.' identifier
                 | identifier

-- WHERE --
<opt_where> ::= ε
                 | WHERE <condition>

<condition> ::= <simple_cond>
                 | <condition> AND <condition>
                 | <condition> OR <condition>
                 | NOT <condition>
                 | '(' <condition> ')'

<simple_cond> ::= <col_ref> <comp_op> <literal>
                 | <col_ref> IN '(' <lit_list> ')'
                 | <col_ref> NOT IN '(' <lit_list> ')'

<comp_op> ::= '=' | '!=' | '>' | '<' | '>=' | '<='

<literal> ::= int_literal | float_literal | string_literal

<lit_list> ::= <literal>
                 | <lit_list> ',' <literal>

-- GROUP BY / ORDER BY / LIMIT --
<opt_group> ::= ε | GROUP BY <col_ref_list>
<opt_order> ::= ε | ORDER BY <col_ref_list>
<opt_limit> ::= ε | LIMIT int_literal

3. Αρχεία Περιγραφής της Γλώσσας

3.1 Αρχείο Flex — mysqlq.l

%{
/*
 * mysqlq.l — Flex lexer for the mySQLq pseudo-SQL language
 *
 * Keywords are case-insensitive.
 * Identifiers are case-sensitive.
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "mysqlq.tab.h"

void yyerror(const char *s);
%}

%option yylineno
%option noyywrap

DIGIT      [0-9]
ID_START   [a-zA-Z_]
ID_CONT    [a-zA-Z0-9_]

%%

"--"[^\n]*              { /* single-line comment — skip */ }

"/*"                    {
    /* multi-line comment */
    int c, prev = 0;
    while ((c = input()) != EOF) {
        if (prev == '*' && c == '/') break;
        prev = c;
    }
    if (c == EOF)
        yyerror("Unterminated block comment");
}

[ \t\r\n]+              { /* whitespace — skip */ }

[Ss][Ee][Ll][Ee][Cc][Tt]           { return SELECT;      }
[Ff][Rr][Oo][Mm]                   { return FROM;        }
[Ww][Hh][Ee][Rr][Ee]               { return WHERE;       }
[Ll][Ii][Mm][Ii][Tt]               { return LIMIT;       }
[Gg][Rr][Oo][Uu][Pp]               { return GROUP;       }
[Oo][Rr][Dd][Ee][Rr]               { return ORDER;       }
[Bb][Yy]                           { return BY;          }
[Aa][Nn][Dd]                       { return AND;         }
[Oo][Rr]                           { return OR;          }
[Nn][Oo][Tt]                       { return NOT;         }
[Cc][Rr][Ee][Aa][Tt][Ee]           { return CREATE;      }
[Tt][Aa][Bb][Ll][Ee]               { return TABLE;       }
[Jj][Oo][Ii][Nn]                   { return JOIN;        }
[Oo][Nn]                           { return ON;          }
[Aa][Ss]                           { return AS;          }
[Ii][Nn][Tt]                       { return INT_TYPE;    }
[Ff][Ll][Oo][Aa][Tt]               { return FLOAT_TYPE;  }
[Vv][Aa][Rr][Cc][Hh][Aa][Rr]       { return VARCHAR;     }
[Ii][Nn]                           { return IN_KW;       }

{ID_START}{ID_CONT}*    { yylval.sval = strdup(yytext); return IDENTIFIER; }

"-"?{DIGIT}+"."{DIGIT}+ { yylval.dval = atof(yytext);  return FLOAT_LIT;  }
"-"?{DIGIT}+             { yylval.ival = atoi(yytext);  return INT_LIT;    }

\'[^\']*\'               { yylval.sval = strdup(yytext); return STRING_LIT; }

"!="  { return NE; }   ">=" { return GE; }   "<=" { return LE; }
">"   { return GT; }   "<"  { return LT; }   "="  { return EQ; }
"."   { return DOT; }  ","  { return COMMA; }   ";"  { return SEMICOLON; }
"("   { return LPAREN; }   ")"  { return RPAREN; }   "*"  { return STAR; }

.     { char msg[64]; snprintf(msg,sizeof(msg),"Unknown character: '%s'",yytext); yyerror(msg); }
%%

3.2 Αρχείο Bison — mysqlq.y (τελικό, εκτεταμένη έκδοση)

Το αρχείο Bison υλοποιεί την πλήρη γραμματική, τους σημασιολογικούς ελέγχους (Ερώτημα 2) και τις επεκτάσεις JOIN/AS (Ερώτημα 3). Το πλήρες αρχείο βρίσκεται στο mysqlq.y του παραδοτέου.

3.3 Αρχείο Πίνακα Συμβόλων — symtable.h / symtable.c

Υλοποιεί την αποθήκευση πινάκων/στηλών στη μνήμη, τη διαχείριση του query context (FROM + JOINs), την ανάλυση aliases και τον έλεγχο συμβατότητας τύπων.

4.3 Ερώτημα 3 — JOIN και Aliases

✅ Παράδειγμα 3α: Έγκυρο JOIN (test6_join_ok.sql)
CREATE TABLE departments ( dept_id int, dept_name varchar(100) );
CREATE TABLE staff ( staff_id int, full_name varchar(100), dept_id int, salary float );

SELECT staff.full_name, departments.dept_name
FROM staff
JOIN departments ON staff.dept_id = departments.dept_id
WHERE staff.salary > 50000;

--- Syntax and semantics OK ---
✅ Παράδειγμα 3β: Έγκυρες Aliases (test7_alias_ok.sql)

FROM customers AS c
JOIN invoices AS i ON c.cust_id = i.cust_id
WHERE i.total > 500.0
ORDER BY c.cust_name
LIMIT 10;

--- Syntax and semantics OK ---

CREATE TABLE /* inline comment */ products (

SELECT prod_id, /* select comment */ prod_name

WHERE price > 10.0 -- end-of-line comment

5. Σχόλια – Παραδοχές

  • Αρνητικοί αριθμοί: Αναγνωρίζονται απευθείας από τον λεκτικό αναλυτή ως ενιαίο token (π.χ. -512, -3.14). Δεν αντιμετωπίζονται ως μοναδιαίος αρνητικός τελεστής.
  • Σχόλια: Υποστηρίζονται σχόλια μίας γραμμής (--) και πολλαπλών γραμμών (/* */). Μπορούν να εμφανιστούν ανάμεσα σε οποιαδήποτε tokens, αλλά δεν επιτρέπεται να σπάνε ένα token.
  • Εκτύπωση προγράμματος: Σε περίπτωση σφάλματος, εκτυπώνονται μόνο οι γραμμές μέχρι και τη γραμμή του σφάλματος. Σε επιτυχή ανάλυση, εκτυπώνεται ολόκληρο το πρόγραμμα.
  • Αναβολή επικύρωσης SELECT στηλών: Οι στήλες του SELECT επικυρώνονται σημασιολογικά μετά την επεξεργασία των όρων FROM και JOIN, καθώς η γραμματική επεξεργάζεται το SELECT πριν το FROM.
  • Σειρά όρων SELECT: Η σειρά SELECT → FROM → WHERE → GROUP BY → ORDER BY → LIMIT είναι αυστηρή. Διαφορετική σειρά αναγνωρίζεται ως συντακτικό σφάλμα.
  • Aliases: Κάθε alias πρέπει να είναι μοναδικό εντός ενός ερωτήματος. Εάν δηλωθεί alias για έναν πίνακα, η αναφορά στις στήλες του γίνεται μέσω του alias.
  • Τύποι δεδομένων: INT είναι συμβατό μόνο με ακέραια κυριολεκτικά, FLOAT με ακέραια ή πραγματικά, VARCHAR με αλφαριθμητικά (single-quoted).
  • Σημασιολογικοί έλεγχοι Q2–Q3 υλοποιούνται αποκλειστικά σε C μέσω του αρχείου symtable.c και ενσωματωμένων actions στο Bison.
